📌 Detect Outliers & Compare Statistics


# Select numeric columns
numeric_cols <- sapply(data, is.numeric)
data_numeric <- data[, numeric_cols]

# Function to detect outliers using IQR
detect_outliers <- function(x) {
  Q1 <- quantile(x, 0.25, na.rm=TRUE)
  Q3 <- quantile(x, 0.75, na.rm=TRUE)
  IQR <- Q3 - Q1
  return(x < (Q1 - 1.5*IQR) | x > (Q3 + 1.5*IQR))
}

# Identify rows with any outliers
outlier_rows <- apply(data_numeric, 1, function(row) any(detect_outliers(row)))

# Compare statistics
stats_before <- data_numeric %>% summarise(across(everything(), list(mean=mean, sd=sd), na.rm=TRUE))
stats_after  <- data_numeric[!outlier_rows, ] %>% summarise(across(everything(), list(mean=mean, sd=sd), na.rm=TRUE))

stats_before
stats_after

# Remove duplicate rows (if any)
data_clean <- data %>% distinct()

# Optional: convert character columns to factors
data_clean <- data_clean %>% mutate(across(where(is.character), as.factor))

📌 Export Cleaned Dataset


write.csv(data_clean, "data_cleaned.csv", row.names = FALSE)

Step 4 — Data Cleaning with dplyr (After Missing Values)

After generating your ~5% missing dataset, download it and use the following R code to handle missing values, detect outliers, normalize numeric columns, and summarize the cleaned data.

📌 Import the Dataset with Missing Values


library(dplyr)

# Replace 'dataset_missing.csv' with your downloaded file
data_missing <- read.csv("dataset_missing.csv")

# View first rows
head(data_missing)

# Check structure
str(data_missing)

# Count missing values
colSums(is.na(data_missing))

📌 Handle Missing Values (Imputation)


# Impute numeric columns with mean, factors/characters with mode
mode_value <- function(x){names(sort(table(x), decreasing = TRUE))[1]}

data_imputed <- data_missing %>%
  mutate(across(where(is.numeric), ~ ifelse(is.na(.), mean(., na.rm = TRUE), .))) %>%
  mutate(across(where(is.character), ~ ifelse(is.na(.), mode_value(.), .)))

# Verify missing values are handled
colSums(is.na(data_imputed))

📌 Outlier Detection (IQR) & Normalization


# Detect outliers using IQR
numeric_cols <- sapply(data_imputed, is.numeric)
outliers <- lapply(data_imputed[, numeric_cols], function(x){
  Q1 <- quantile(x, 0.25)
  Q3 <- quantile(x, 0.75)
  IQR <- Q3 - Q1
  x < (Q1 - 1.5*IQR) | x > (Q3 + 1.5*IQR)
})

# Optionally remove or flag outliers
# data_clean <- data_imputed[!rowSums(do.call(cbind, outliers)), ]

# Min-max normalization
normalized_data <- data_imputed %>%
  mutate(across(where(is.numeric), ~ (.-min(.))/(max(.)-min(.))))

# Summary of numeric variables
summary_stats <- normalized_data %>%
  summarise(across(where(is.numeric), list(mean = mean, median = median, sd = sd)))
summary_stats

📌 Export Cleaned Dataset


write.csv(normalized_data, "dataset_cleaned.csv", row.names = FALSE)
